# Cloud network groups

set firewall group network-group CLOUD-APP network 10.20.10.0/24
set firewall group network-group CLOUD-DB network 10.20.20.0/24

# Rule 5 - Admin denied to Cloud DB
set firewall ipv4 forward filter rule 5 action drop
set firewall ipv4 forward filter rule 5 description "ADMIN denied to CLOUD-DB"
set firewall ipv4 forward filter rule 5 source group network-group ADMIN
set firewall ipv4 forward filter rule 5 destination group network-group CLOUD-DB
set firewall ipv4 forward filter rule 5 log

# Rule 6 - Admin allowed to Cloud App
set firewall ipv4 forward filter rule 6 action accept
set firewall ipv4 forward filter rule 6 description "ADMIN permitted to CLOUD-APP"
set firewall ipv4 forward filter rule 6 source group network-group ADMIN
set firewall ipv4 forward filter rule 6 destination group network-group CLOUD-APP

# Rule 7 - Design denied from AWS private networks
set firewall ipv4 forward filter rule 7 action drop
set firewall ipv4 forward filter rule 7 description "DESIGN denied all cloud private"
set firewall ipv4 forward filter rule 7 source group network-group DESIGN
set firewall ipv4 forward filter rule 7 destination address 10.20.0.0/16
set firewall ipv4 forward filter rule 7 log
